# ── 맨손 mode 1 + Loop ──
# 터미널 1
bash ~/Downloads/external_repos/ai_worker_teleop_deploy/teleop.sh robot \
  --pc-ip 192.0.2.10 --robot-ip 192.0.2.20 \
  --loop vr --input hands --hand-mode 1 --bringup

# 터미널 2 (PC에서 Loop 앱을 먼저 열기)
bash ~/Downloads/external_repos/ai_worker_teleop_deploy/teleop.sh pc \
  --pc-ip 192.0.2.10 --robot-ip 192.0.2.20 \
  --loop vr --input hands --hand-mode 1

# Quest
https://192.0.2.10:8012?ws=wss://192.0.2.10:8012
# Loop 소스: aiworker / HX5-D20
# Loop 창: V=시작·HOLD·재개, WASD=이동, O/P=회전, 방향키=목, U/I=lift
# 이동 키를 놓은 뒤 V 한 번으로 재개. C=Loop 검토, Shift+A=녹화.
# 종료: Ctrl+C. robot 터미널의 정리·손 토크 OFF 완료까지 대기.

# ============================================================
# 디버깅
# ============================================================
# 일반 ROS 로그 (teleop 실행 중, 세 번째 PC 터미널)
ssh -t robotis@192.0.2.20 \
  'docker exec -it ai_worker bash /workspace/ai_worker_xr/hardware/log_hands.sh --seconds 120'

# 드라이버 상세 기록: hardware/HAND_TRACE.md
# 통신 지연 조사: hardware/HX5_LATENCY.md
# X11 권한은 teleop.sh pc가 자동 설정. 수동 확인이 필요할 때만:
xhost +SI:localuser:root

# ============================================================
# 설정 / 업데이트
# ============================================================
# 새 PC 최초 설치
mkdir -p ~/Downloads/external_repos
cd ~/Downloads/external_repos
git clone --branch main --single-branch \
  https://github.com/MFIWO/ai_worker_teleop.git ai_worker_teleop_deploy
cd ai_worker_teleop_deploy
bash docker/container.sh start
bash hardware/install_pc.sh
# 실제 PC IP로 인증서 생성; 새 CA는 Quest에서 신뢰 설정
PC_IP=192.0.2.10
docker exec -it robotis-applications bash /root/gen_cert.sh "$PC_IP"

# 기존 PC 업데이트 (먼저 양쪽 teleop 종료)
cd ~/Downloads/external_repos/ai_worker_teleop_deploy
git switch main
git pull --ff-only origin main
bash hardware/install_pc.sh

# PC에서 로봇으로 배포 (최초 설치 / 업데이트 공통)
ROBOT_IP=192.0.2.20
cd ~/Downloads/external_repos/ai_worker_teleop_deploy
rsync -av \
  --exclude='.git/' --exclude='__pycache__/' --exclude='*.pyc' \
  --exclude='*.pem' --exclude='*.key' --exclude='docker/workspace/' \
  ./ "robotis@$ROBOT_IP:~/ai_worker_teleop_sh5/"
ssh "robotis@$ROBOT_IP" \
  'cd ~/ai_worker_teleop_sh5 && bash hardware/install_robot.sh'
ssh "robotis@$ROBOT_IP" 'docker exec -i ai_worker bash -s' <<'ROBOT_BUILD'
set -e
source /opt/ros/jazzy/setup.bash
source /root/ros2_ws/install/setup.bash
export PYTHONPATH="/opt/venv/lib/python3.12/site-packages:${PYTHONPATH:-}"
cd /workspace/ai_worker_xr
python3 -m hardware.build_sh5_hands --verify-only || \
  python3 -m hardware.build_sh5_hands --rebuild
python3 -m hardware.build_hand_write_rate
ROBOT_BUILD

# 선택: 상세 드라이버 기록용 별도 빌드 (일반 실행에는 불필요)
ssh "robotis@$ROBOT_IP" 'docker exec -i ai_worker bash -s' <<'TRACE_BUILD'
set -e
source /opt/ros/jazzy/setup.bash
source /root/ros2_ws/install/setup.bash
cd /workspace/ai_worker_xr
python3 -m hardware.build_hand_trace
TRACE_BUILD
