# ============================================================

# ============================================================
# 4. AI Worker 컨테이너에서 C++ 빌드 + manifest 검증
# ============================================================
ssh "robotis@$ROBOT_IP" 'docker exec -i ai_worker bash -s' <<'ROBOT_BUILD'
set -eo pipefail
source /opt/ros/jazzy/setup.bash
source /root/ros2_ws/install/setup.bash
export PYTHONPATH="/workspace/ai_worker_xr/vendor:/workspace/ai_worker_xr:/opt/venv/lib/python3.12/site-packages:${PYTHONPATH:-}"
cd /workspace/ai_worker_xr
# 실행 중인 follower가 있으면 중단하고 종료부터 진행합니다.
if pgrep -f '[r]os2_control_node' >/dev/null; then
  echo '[STOP] ros2_control_node 실행 중입니다. robot 세션을 먼저 종료하세요.' >&2
  exit 1
fi
# 팔 solver/팔꿈치 제한/충돌 모델 변경은 반드시 C++ 재빌드가 필요합니다.
if ! python3 -B -m hardware.build_sh5_hands --verify-only; then
  python3 -B -m hardware.build_sh5_hands --rebuild
fi
python3 -B -m hardware.build_sh5_hands --verify-only
# 손 쓰기 30Hz + 공유 읽기 timeout cap 60ms 드라이버.
# manifest 누락/이전 버전은 기존 빌드 폴더를 백업한 뒤 새로 빌드합니다.
if ! python3 -B -m hardware.build_hand_write_rate --verify-only; then
  build_root=/workspace/ai_worker_xr_hand_write_30hz_bounded_v1
  if [ -e "$build_root" ]; then
    mv "$build_root" "${build_root}.backup.$(date +%Y%m%d_%H%M%S_%N)"
  fi
  python3 -B -m hardware.build_hand_write_rate
fi
python3 -B -m hardware.build_hand_write_rate --verify-only
# 공식 ROBOTIS 촉각 broadcaster만 별도 빌드. 모터/별도 손 드라이버 실행 없음.
python3 -B -m hardware.build_hand_pressure
python3 -B -m hardware.build_hand_pressure --verify-only
ROBOT_BUILD
# 두 빌드가 Verified여야 실행합니다. 전체 AI Worker colcon 재빌드는 불필요합니다.
# 새 프로필은 다음 robot 세션이 생성·검증합니다. 예전 프로필을 수동 복사하지 마세요.
# 소스 hash가 다르다는 오류는 robot 기반 Cyclo/드라이버 버전을 확인해야 합니다.

# ============================================================
# 5. 맨손 mode 1 + Loop — PC 호스트 터미널 두 개
# ============================================================
# 먼저 PC Loop 앱을 엽니다. 각 터미널에서 PC_IP/ROBOT_IP/TELEOP_REPO를 설정하세요.
# 터미널 1: 로봇 세션. --bringup은 lift를 물리 상단으로 올린 뒤 팔을 5초 준비 자세 이동; 손 초기화 궤적 없음.
bash "$TELEOP_REPO/teleop.sh" robot \
  --pc-ip "$PC_IP" --robot-ip "$ROBOT_IP" \
  --loop vr --input hands --hand-mode 1 --bringup
# 터미널 2: PC 세션
bash "$TELEOP_REPO/teleop.sh" pc \
  --pc-ip "$PC_IP" --robot-ip "$ROBOT_IP" \
  --loop vr --input hands --hand-mode 1
# Quest: https://PC_IP:8012?ws=wss://PC_IP:8012 (PC_IP를 실제 주소로 치환)
# Loop 소스: aiworker / HX5-D20
# Loop 창: V=시작·HOLD·재개, WASD=이동, O/P=회전, 방향키=목, U/I=lift.
# Shift+A=녹화. C=검토. 이동 키를 놓은 뒤 V로 재앵커·재개.

# ============================================================
# 6. 컨트롤러 mode 2 + Loop — PC 호스트 터미널 두 개
# ============================================================
# 터미널 1
bash "$TELEOP_REPO/teleop.sh" robot \
  --pc-ip "$PC_IP" --robot-ip "$ROBOT_IP" \
  --loop vr --input controllers --hand-mode 2 --bringup
# 터미널 2
bash "$TELEOP_REPO/teleop.sh" pc \
  --pc-ip "$PC_IP" --robot-ip "$ROBOT_IP" \
  --loop vr --input controllers --hand-mode 2
# 양쪽 --input/--hand-mode를 동일하게 맞춥니다.
# Loop 없이 사용: 양쪽 모두 --loop off. 다른 mode는 양쪽 --hand-mode를 함께 변경.
# A+X: 시작·재개. [XR RESUME READY]까지 컨트롤러를 가만히 유지.
# B+Y: HOLD. mode 2는 마지막 손 목표를 유지.
# 왼쪽 grip: 누르는 동안 왼팔 추종, 놓으면 유지. 오른팔은 활성화 후 추종.
# 오른쪽 grip: 누르는 동안 오른팔 추종, 놓으면 유지. 다시 누르면 현재 입력에서 재개.
# B 단독: 팔 준비 자세 복귀 요청. 완료 로그 후 A+X로 수동 재개.
# A 단독: pointing. Trigger에 따라 검지 유지 전류 한도 증가(기존 300 raw 이내).
# X 단독: 원래 mode 1 grip pinch 궤적. 엄지 25%, 검지·중지 15% 굽힌 상태에서 시작.
# Y 단독: 엄지·검지만 pinch. 중지·약지·새끼는 열린 상태 유지.
# Trigger를 갑자기 놓아도 손은 속도 제한을 따라 천천히 열림.
# 팔꿈치 QP 가중치는 정지/이동 모두 0. 이동 중에도 grip을 누른 팔은 추종.
# 오른쪽 stick: lift + MOBILE 회전을 대각선으로 동시에 입력 가능.
# 양쪽 stick 클릭: HEAD/MOBILE 전환. 왼쪽 head/mobile 축 선택 정책 유지.
# 변경된 Y pose JSON은 robot 설치 + 세션 재시작으로 반영(C++ 재빌드 불필요).
# 이번 전체 버전 업데이트는 4번 SH5 C++ 빌드도 수행해야 합니다.
# robot 명령에 --base-speed 0.10 --turn-speed 0.15 추가 가능.
# --base-speed: 전후/좌우 최대 이동 속도 m/s (기본 0.20, 범위 0..0.20).
# --turn-speed: 좌우 회전 최대 속도 rad/s (기본 1/3, 범위 0..1/3).
# PC 명령은 그대로 사용. 속도 선택은 robot gate에서 적용되며 C++ 재빌드 불필요.
# grip 해제 / B+Y HOLD 중에도 Loop 데이터는 30Hz 연속 전송.
# HOLD action은 측정 위치로 기록하며 action_valid=0. 로봇에 추가 명령을 보내지 않음.
# 첫 프레임부터 기록 열 고정: lift 명령이 뒤늦게 생겨 Parquet 저장 실패하던 오류 수정.
# 기본 training: 19채널(관절 관측/명령 12 + flags/valid 3 + 양손 pressure/status 4).
# robot-aiworker.parquet에 observation.left_hand.tactile_pressure / right_hand.tactile_pressure 포함.
# pressure는 손마다 45개. 누락/오래된 데이터는 NaN, tactile_status의 fresh=0으로 구분.
# 변경 후 Loop Preview에서 aiworker를 다시 선택하고 Cell Config 저장 후 새 recording 시작.

# 지연 조사 원본은 과거 기록이며, 2026-10-29 사용자가 현재 환경 지연 해결을 확인했습니다.

# 추가 수정 반영: PC install + robot 소스 재배포 후 SH5 C++ --rebuild 필요.
# HX5-D20 촉각 소스는 양손 실제 압력 데이터가 들어온 뒤에만 Loop에 등록됨.
# HandPressures broadcaster가 없으면 /dynamic_joint_states의 센서 인터페이스를 사용.
# --loop vr 실행 시 공식 pressure_l/r_broadcaster를 기존 SH5 controller_manager에 추가.
# 처음에는 follower를 종료 후 --bringup으로 재시작해야 촉각 plugin overlay가 반영됨.
# hx5_d20_right.launch.py를 별도로 실행하지 않음(손 포트/하드웨어 드라이버 중복 방지).
# 확인: ros2 control list_controllers / ros2 topic echo /right_hand/finger_pressures --once
# 양손 각각 sensor1..5 = 엄지/검지/중지/약지/새끼, 손가락마다 압력 9개(0..255).
# 무접촉에서 0은 정상. 오른손 엄지 ID115 눌렀을 때 255까지 반응하고 떼면 0 확인.
# ROS uint8 배열(numpy.uint8)을 거부하던 파서도 수정. 실제 ROS 메시지 회귀검사 통과.
